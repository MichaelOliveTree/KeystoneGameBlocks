Memory<T> as blackboard data

// Example using Memory<T> for shared data in a blackboard scenario
// NOTE: Here, instead of using string "key" values and a dictionary, we use
// explicit offsets into the byte[] array to read the data types we want to read.
// This data then needs to be cast to the appropriate Type.
// Storing "objects" is really not an ideal use case here...  but for "Policies"
// like Directives, RulesOfEngagement, it can be a very good idea.
// If we wanted to, we could use a Dictionary to store the offsets for every
// bit of data we wanted to access...  We would also know the type of data
// but if not, for an array for instance, we could store a length as well.

public class BlackboardManager
{
    // The shared memory area
    private Memory<byte> _sharedMemory;

    public BlackboardManager(int size)
    {
        // Allocate shared memory
        _sharedMemory = new byte[size].AsMemory();
    }

    // Agent reads data
    public ReadOnlySpan<byte> ReadData(int start, int length)
    {
        return _sharedMemory.Slice(start, length).Span;
    }

    // Agent updates data
    public void WriteData(int start, byte[] data)
    {
        data.CopyTo(_sharedMemory.Slice(start, data.Length).Span);
    }
}
